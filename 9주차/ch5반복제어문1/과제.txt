// 5번 문제
#include <stdio.h>

int main() {
    int score;
    while (1) {
        scanf("%d", &score);
        if (score > 100 || score < 0) {
            break;
        }
        if (score >= 80) {
            printf("Pass");
        }
        else {
            printf("Fail");
        }
    }
}

// 7번 문제
#include <stdio.h>

int main() {
    int score = 0;
    int score_check;
    int loop = 0;
    while (1) {
        scanf("%d", &score_check);
        if (score_check == 0) {
            break;
        }
        
        score += score_check;
        loop++;
    }
    printf("cnt = %d\nsum = %d\navg = %2.f", loop, score,  (double)score/loop);
}
