    scanf("%d", &n);

    for (int i = 0; i < n; i++) {
        scanf("%d", &score);
        sum += score;
    }

    avg = (double)sum / n;

    } else {
        printf("fail\n");
    }

    printf("%.1f\n", avg);

    return 0;
}

// 22번 문제
#include <stdio.h>

int main(void)
{
    int A, B;
    int sum = 0;
    int count = 0;

    scanf("%d %d", &A, &B);

    for (int i = A; i <= B; i++)
    {
        if (i % 3 == 0 || i % 5 == 0)
        {
            sum += i;
            count++;
        }
    }

    double avg = (double)sum / count;

    printf("%d\n", sum);
    printf("%.3f\n", avg);

    return 0;
}

// 24번 문제
#include <stdio.h>

int main(void)
{
    int row, col;

    scanf("%d %d", &row, &col);

    for (int i = 1; i <= row; i++)
    {
        for (int j = 1; j <= col; j++)
        {
            printf("%d", i * j);

            if (j < col)
            {
                printf(" ");
            }
        }
        printf("\n");
    }

    return 0;
}
