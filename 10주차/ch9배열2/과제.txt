// 1번 문제
#include <stdio.h>

int main() {
    int arr[11] = {0};
    int num;

    while (1) {
        scanf("%d", &num);

        if (num < 1 || num > 10) {
            break;
        }

        arr[num]++;
    }

    for (int i = 1; i <= 10; i++) {
        if (arr[i] > 0) {
            printf("%d : %d\n", i, arr[i]);
        }
    }

    return 0;
}

// 2번 문제
#include <stdio.h>

int main() {
    int arr[26] = {0};
    char ch;

    while (1) {
        scanf(" %c", &ch);

        if (ch < 'A' || ch > 'Z') {
            break;
        }

        arr[ch - 'A']++;
    }

    for (int i = 0; i < 26; i++) {
        if (arr[i] > 0) {
            printf("%c : %d\n", 'A' + i, arr[i]);
        }
    }

    return 0;
}

// 11번 문제
#include <stdio.h>

int main() {
    int score[3][3];
    int studentSum, subjectSum[3] = {0};
    int totalSum = 0;

    for (int i = 0; i < 3; i++) {
        for (int j = 0; j < 3; j++) {
            scanf("%d", &score[i][j]);
        }
    }

    for (int i = 0; i < 3; i++) {
        studentSum = 0;

        for (int j = 0; j < 3; j++) {
            printf("%d ", score[i][j]);

            studentSum += score[i][j];
            subjectSum[j] += score[i][j];
        }

        printf("%d\n", studentSum);
        totalSum += studentSum;
    }

    for (int i = 0; i < 3; i++) {
        printf("%d ", subjectSum[i]);
    }

    printf("%d\n", totalSum);

    return 0;
}urn 0;
}

    }

    for (i = 0; i < n - 1; i++) {

// 22번 문제
#include <stdio.h>

int main() {
    int n;
    int arr[100][100] = {0};

    scanf("%d", &n);

    for (int i = 0; i < n; i++) {
        arr[i][0] = 1;
        arr[i][i] = 1;
    }

    for (int i = 2; i < n; i++) {
        for (int j = 1; j < i; j++) {
            arr[i][j] = arr[i - 1][j - 1] + arr[i - 1][j];
        }
    }

    for (int i = n - 1; i >= 0; i--) {
        for (int j = 0; j <= i; j++) {
            printf("%d ", arr[i][j]);
        }
        printf("\n");
    }

    return 0;
}
