// 3번 문제
#include <stdio.h>

int main() {
    int i, j;

    for (i = 1; i <= 6; i++) {
        int stars;

        if (i <= 3) {
            stars = i;
        } else {
            stars = 6 - i;
        }

        for (j = 1; j <= stars; j++) {
            printf("*");
        }

        printf("\n");
    }

    return 0;
}

// 11번 문제
#include <stdio.h>

int main() {
    int i, j;
    int num = 1;

    for (i = 0; i < 4; i++) {
        for (j = 0; j < 4; j++) {
            if (j == 0) {
                printf("%c ", 'a' + i);
            } else if (i == 0) {
                printf("%d ", num);
                num++;
            } else {
                printf("%c ", 'a' + i + j);
                num++;
            }
        }
        printf("\n");
    }

    return 0;
}

// 12번 문제
#include <stdio.h>

int main() {
    int n;
    int i, j;
    int num = 1;

    scanf("%d", &n);

    for (i = 0; i < n; i++) {
        for (j = 0; j < n + 1; j++) {
            if (j < n - i) {
                printf("%d ", num);
                num++;
            } else {
                printf("%c ", 'A' + j - (n - i));
            }
        }
        printf("\n");
    }

    return 0;
}

// 14번 문제
#include <stdio.h>

int main() {
    int n;
    int i, j;

    scanf("%d", &n);

    for (i = 0; i < n; i++) {
        for (j = 0; j < 2 * n - 1; j++) {
            if (j >= i && j < 2 * n - 1 - i) {
                printf("*");
            } else {
                printf(" ");
            }
        }
        printf("\n");
    }

    for (i = n - 2; i >= 0; i--) {
        for (j = 0; j < 2 * n - 1; j++) {
            if (j >= i && j < 2 * n - 1 - i) {
                printf("*");
            } else {
                printf(" ");
            }
        }
        printf("\n");
    }

    return 0;
}

// 17번 문제
#include <stdio.h>

int main() {
    int n;
    scanf("%d", &n);

    char ch = 'A';
    int num = 0;

    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            if (i + j < n) {
                printf("%c ", ch++);
            }
            else {
                printf("%d ", num++);
            }
        }
        printf("\n");
    }

    return 0;
}

    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= i; j++) {
            printf("# ");
        }
        printf("\n");
    }

    for (int i = n - 1; i >= 1; i--) {
        for (int j = 0; j < n - i; j++) {
            printf("  ");
        }
        for (int j = 1; j <= i; j++) {
            printf("# ");
        }
        printf("\n");
    }

    return 0;
}
