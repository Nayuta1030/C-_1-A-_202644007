// 7번 문제
#include <stdio.h>

int main() {
    int score = 0;
    int score_check;
    int loop = 0;
    while (1) {
        scanf("%d", &score_check);
        if (score_check == 0) {
            break;
        }

        score += score_check;
        loop++;
    }
    if (loop > 0) {
        printf("cnt = %d\nsum = %d\navg = %.2f", loop, score, (double)score / loop);
    }
}

// 9번 문제

    int num;
    int max = 0;
    int min = 101;

    while (1) {
        scanf("%d", &num);

        if (num == 0) {
            break;
        }

        if (num >= 1 && num <= 100) {
            if (num > max) {
                max = num;
            }

            if (num < min) {
                min = num;
            }
        }
    }

    printf("MAX = %d\nMIN = %d", max, min);
}

// 17번 문제
#include <stdio.h>

int main(void) {
    int n;
    int sum = 0;
    int count = 0;

    while (1) {
        scanf("%d", &n);

        if (n < 0 || n > 100) {
            break;
        }

        sum += n;
        count++;
    }

    printf("sum : %d\n", sum);
    printf("avg : %.1f\n", (double)sum / count);

    return 0;
}

    int x = 50;         // 초기값
    cin >> x;            // [A] 이 줄에 커서가 있을 때 x 값 확인

    if (x >= 80) {      
        x += 10;        // [B] 이 줄에 커서가 있을 때 x 값 확인
    } else {           
        x -= 5;         // [B] 이 줄에 커서가 있을 때 x 값 확인
    }
    x--;                // [C] 이 줄에 커서가 있을 때 x 값 확인

    int A = 75;
    int B = 75;
    int C = 70;         // [C]에서 확인한 값으로 수정

    cout << A << " " << B << " " << C << '\n';

}
