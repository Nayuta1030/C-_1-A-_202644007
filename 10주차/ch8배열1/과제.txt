// 20번 문제
#include <stdio.h>

int main() {
    int arr[100];
    int n = 0;
    int count = 0;
    int sum = 0;
    double avg;

    while (n < 100) {
        scanf("%d", &arr[n]);

        if (arr[n] == 0) {
            break;
        }

        n++;
    }

    for (int i = 0; i < n; i++) {
        if (arr[i] % 5 == 0) {
            count++;
            sum += arr[i];
        }
    }

    if (count > 0) {
        avg = (double)sum / count;
    }
    else {
        avg = 0;
    }

    printf("%d\n", count);
    printf("%d\n", sum);
    printf("%.1f", avg);

    return 0;
}
