// 1번 문제
#include <stdio.h>

int main() {
    for (int i = 1; i < 4; i++) {
        for (int j = 0; j < i; j++) {
            printf("*");
        }
        printf("\n");
    }
    return 0;
}

// 2번 문제
#include <stdio.h>

int main() {
    int a;
    scanf("%d", &a);

    for (int i = 1; i <= a; i++) {
        for (int j = 0; j < i; j++) {
            printf("⁕");
        }
        printf("\n");
    }
    return 0;
}
